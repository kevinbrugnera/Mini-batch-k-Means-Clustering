# Spark configurations

1. Topology Benchmarks

- Thin:
spark-submit \
  --master $MASTER_URL \
   --driver-memory 2G \
   --executor-memory 1G \
   --conf spark.executor.memoryOverhead=750 \
   --conf spark.memory.fraction=0.8 \
   --conf spark.memory.storageFraction=0.7 \
   --total-executor-cores 16 \
   --executor-cores 1 \
   --conf spark.default.parallelism=$NUM_PARTITIONS \
   --py-files functions.py \
   $TARGET_SCRIPT

- Balanced:
spark-submit \
  --master spark://$MASTER_IP:7077 \
  --driver-memory 2G \
  --executor-memory 2500M \
  --conf spark.executor.memoryOverhead=1000 \
  --conf spark.memory.fraction=0.8 \
  --conf spark.memory.storageFraction=0.7 \
  --total-executor-cores 16 \
  --executor-cores 2 \
  --conf spark.default.parallelism=48 \
  --py-files functions.py \
  $TARGET_SCRIPT

- Fat:
